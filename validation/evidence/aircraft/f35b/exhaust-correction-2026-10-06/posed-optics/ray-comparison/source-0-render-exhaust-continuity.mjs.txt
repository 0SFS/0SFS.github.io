const add = (a, b) => a.map((v, i) => v + b[i]), sub = (a, b) => a.map((v, i) => v - b[i]), scale = (a, s) => a.map(v => v * s), dot = (a, b) => a.reduce((n, v, i) => n + v * b[i], 0);

function slab(origin, direction, lo, hi) { let begin = 0, end = Infinity; for (let c = 0; c < 3; c++) { if (Math.abs(direction[c]) < 1e-14) { if (origin[c] < lo[c] || origin[c] > hi[c]) return null; continue; } const a = (lo[c] - origin[c]) / direction[c], b = (hi[c] - origin[c]) / direction[c]; begin = Math.max(begin, Math.min(a, b)); end = Math.min(end, Math.max(a, b)); } return end > begin ? [begin, end] : null; }
function hardwareDistance(origin, direction, tree) {
  if (!slab(origin, direction, tree.min, tree.max)) return Infinity;
  if (tree.children) return Math.min(...tree.children.map(child => hardwareDistance(origin, direction, child)));
  let closest = Infinity; const end = add(origin, scale(direction, 40));
  for (const item of tree.triangles) { const hit = segmentTriangle(origin, end, item); if (hit) closest = Math.min(closest, Math.hypot(...sub(hit, origin))); }
  return closest;
}
function prepareSections(support) {
  return support.sections.map(s => {
    const d = sub(s.endCenter, s.startCenter), du = sub(s.endU, s.startU), min = [Infinity, Infinity, Infinity], max = [-Infinity, -Infinity, -Infinity];
    for (const [center, u, radius] of [[s.startCenter, s.startU, s.startRadius], [s.endCenter, s.endU, s.endRadius]]) for (let i = 0; i < 3; i++) { const e = radius * Math.hypot(u[i], s.radialV[i]); min[i] = Math.min(min[i], center[i] - e); max[i] = Math.max(max[i], center[i] + e); }
    return { ...s, d, du, min, max, a: -dot(cross(d, du), s.radialV) };
  });
}
// Independent scalar inverse; source interpolation below also does not call the
// production sampler. Exact triangle occlusion clips the ray before integration.
function localCoordinate(sections, point) {
  for (let index = 0; index < sections.length; index++) {
    const s = sections[index]; if (point.some((v, i) => v < s.min[i] - 1e-6 || v > s.max[i] + 1e-6)) continue;
    const p = sub(point, s.startCenter), b = dot(cross(p, s.du), s.radialV) - dot(cross(s.d, s.startU), s.radialV), c = dot(cross(p, s.startU), s.radialV);
    let roots;
    if (Math.abs(s.a) < 1e-12) roots = [-c / b];
    else { const det = b * b - 4 * s.a * c; if (det < 0) continue; roots = [(-b + Math.sqrt(det)) / (2 * s.a), (-b - Math.sqrt(det)) / (2 * s.a)]; }
    for (const fraction of roots) {
      if (fraction < -1e-6 || fraction > 1 + 1e-6) continue;
      const t = Math.max(0, Math.min(1, fraction)), u = add(s.startU, scale(s.du, t)), relative = sub(p, scale(s.d, t));
      const x = dot(relative, u) / dot(u, u), y = dot(relative, s.radialV), r = Math.hypot(x, y), radius = s.startRadius + t * (s.endRadius - s.startRadius);
      if (r > radius) continue;
      let inner = 0;
      for (let k = s.innerRadiusKnots.length - 1; k >= 0; k--) if (t >= s.innerRadiusKnots[k][0]) { const a = s.innerRadiusKnots[k], b = s.innerRadiusKnots[k + 1]; inner = b ? a[1] + (b[1] - a[1]) * (t - a[0]) / (b[0] - a[0]) : a[1]; break; }
      if (r < inner) continue;
      if (s.nozzleSealClip) { const half = Math.PI / g.aperture.segmentCount, seal = (radius - g.aperture.sealHalfWidth * Math.sin(half)) / Math.cos(half), angle = Math.atan2(y, x) - half, near = angle - 2 * half * Math.round(angle / (2 * half)); if (r * Math.cos(near) > seal) continue; }
      return { distance: s.startDistance + t * (s.endDistance - s.startDistance), radial: r / radius };
    }
  }
  return null;
}
function bilinear(field, u, v) {
  if (u < 0 || u > 1 || v < 0 || v > 1) return [0, 0, 0, 0];
  const x = u * (field.width - 1), y = v * (field.height - 1), i = Math.min(field.width - 2, Math.floor(x)), j = Math.min(field.height - 2, Math.floor(y)), a = x - i, b = y - j, result = [0, 0, 0, 0];
  for (const [dx, dy, weight] of [[0, 0, (1 - a) * (1 - b)], [1, 0, a * (1 - b)], [0, 1, (1 - a) * b], [1, 1, a * b]]) for (let c = 0; c < 4; c++) result[c] += weight * field.rgba[((j + dy) * field.width + i + dx) * 4 + c]; return result;
}
function integrate(origin, direction, support, sections, gas, opaqueDistance, samples, baseline = false) {
  const interval = slab(origin, direction, support.bounds.min, support.bounds.max), radiance = [0, 0, 0], insideRadiance = [0, 0, 0], outsideRadiance = [0, 0, 0];
  if (!interval || !gas.field) return { radiance, insideRadiance, outsideRadiance, transmittance: 1 };
  const begin = interval[0], end = Math.min(interval[1], opaqueDistance); if (end <= begin) return { radiance, insideRadiance, outsideRadiance, transmittance: 1 };
  const step = (end - begin) / samples, range = gas.field.axialDistanceRangeMeters ?? [0, 6]; let transmittance = 1;
  for (let i = 0; i < samples; i++) {
    const coordinate = localCoordinate(sections, add(origin, scale(direction, begin + (i + .5) * step))); if (!coordinate || baseline && coordinate.distance < 0) continue;
    const local = bilinear(gas.field, (coordinate.distance - range[0]) / (range[1] - range[0]), coordinate.radial), factor = local[3] > 0 ? -Math.expm1(-local[3] * step) / local[3] : step;
    for (let c = 0; c < 3; c++) { const value = transmittance * local[c] * factor; radiance[c] += value; (coordinate.distance < 0 ? insideRadiance : outsideRadiance)[c] += value; } transmittance *= Math.exp(-local[3] * step);
  }
  return { radiance, insideRadiance, outsideRadiance, transmittance };
}

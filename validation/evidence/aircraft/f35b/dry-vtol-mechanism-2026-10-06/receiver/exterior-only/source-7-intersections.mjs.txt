import { sub,dot,cross,add,mul,unit } from './source.mjs';
export function triangle(points,info){return{points,info,min:[0,1,2].map(k=>Math.min(...points.map(p=>p[k]))),max:[0,1,2].map(k=>Math.max(...points.map(p=>p[k])))};}
function overlaps(a,b,epsilon=1e-6){return a.min.every((v,i)=>v<b.max[i]-epsilon&&a.max[i]>b.min[i]+epsilon);}
export function bvh(triangles){const node={min:[0,1,2].map(i=>Math.min(...triangles.map(t=>t.min[i]))),max:[0,1,2].map(i=>Math.max(...triangles.map(t=>t.max[i])))};if(triangles.length<=16){node.triangles=triangles;return node;}const axis=node.max.map((v,i)=>v-node.min[i]).indexOf(Math.max(...node.max.map((v,i)=>v-node.min[i])));triangles.sort((a,b)=>a.min[axis]+a.max[axis]-b.min[axis]-b.max[axis]);const middle=Math.floor(triangles.length/2);node.children=[bvh(triangles.slice(0,middle)),bvh(triangles.slice(middle))];return node;}
export function segmentTriangle(a,b,t){const [p,q,r]=t.points,d=sub(b,a),e1=sub(q,p),e2=sub(r,p),h=cross(d,e2),det=dot(e1,h);if(Math.abs(det)<1e-12)return null;const inv=1/det,s=sub(a,p),u=inv*dot(s,h);if(u<1e-7||u>1-1e-7)return null;const c=cross(s,e1),v=inv*dot(d,c);if(v<1e-7||u+v>1-1e-7)return null;const distance=inv*dot(e2,c);if(distance<=1e-6||distance>=1-1e-6)return null;return add(a,mul(d,distance));}
export function intersections(t,tree,result=[]){if(!overlaps(t,tree))return result;if(tree.children){for(const c of tree.children)intersections(t,c,result);return result;}for(const other of tree.triangles){if(!overlaps(t,other))continue;let point=null;for(let i=0;i<3&&!point;i++)point=segmentTriangle(t.points[i],t.points[(i+1)%3],other)||segmentTriangle(other.points[i],other.points[(i+1)%3],t);if(point){const normal=unit(cross(sub(other.points[1],other.points[0]),sub(other.points[2],other.points[0]))),distances=t.points.map(p=>dot(sub(p,other.points[0]),normal));result.push({a:t.info,b:other.info,point,planeStraddleDepthMeters:Math.min(Math.max(...distances),-Math.min(...distances))});}}return result;}

function fixture(overrides: Partial<EngineExhaustOptions> = {}, webGpu = false, withHotHardware = false, nativeFiltering = false) {
  const engine = new NullEngine();
  engine.getCaps().textureFloatLinearFiltering = nativeFiltering;
  engines.push(engine);
  if (webGpu) Object.defineProperty(engine, "isWebGPU", { value: true });
  const scene = new Scene(engine);
  scene.useRightHandedSystem = true;
  const camera = new FreeCamera("camera", new Vector3(0, 0, 20), scene);
  scene.activeCamera = camera;
  camera.computeWorldMatrix();
  const attachment = new TransformNode("vectoring-nozzle", scene);
  const requestRender = vi.fn();
  const texture = RawTexture.CreateRGBATexture(new Uint8Array(2 * 4 * 4).fill(255), 2, 4, scene, false, false);
  const options: EngineExhaustOptions = {
    attachment, exitPosition: [0, 0, 1], direction: [0, 0, 1], closedRadiusMeters: 0.4,
    openRadiusMeters: 0.6, lengthMeters: 6, opticalProfile: profile, settings, requestRender,
    createTexture: () => texture, whenReady: async () => {}, ...overrides,
  };
  const liner = withHotHardware ? MeshBuilder.CreateCylinder("nozzle-liner", {}, scene) : null;
  const hardwareMaterial = withHotHardware ? new PBRMaterial("nozzle-metal", scene) : null;
  if (liner && hardwareMaterial) {
    liner.parent = options.attachment;
    hardwareMaterial.emissiveColor = new Color3(0.001, 0.0002, 0.00003);
    liner.material = hardwareMaterial;
  }
  const handle = createEngineExhaust(scene, options);
  const material = handle.mesh.material as ShaderMaterial;
  return { scene, camera, attachment: options.attachment, requestRender, texture, handle, material, liner, hardwareMaterial };
}

function thermalFixture() {
  const t = fixture({
    hotSurfaceMaterialNames: ["nozzle-metal"],
    opticalProfile: {
      ...profile,
      surfaceEmission: { temperatureKelvinRange: [600, 1400], unit: "cd/m2", samples: [[0, 0, 0], [0.4, 0.1, 0.01]] },
      temporalEmission: { periodSeconds: 1, samples: [1, 1.04, 1, 0.96] },
    },
  }, false, true);
  return {
    ...t,
    emission: () => (t.liner!.material as PBRMaterial).emissiveColor.clone(),
    restEmission: t.hardwareMaterial!.emissiveColor.clone(),
  };
}

function materialData(material: ShaderMaterial): {
  floats: Record<string, number>; ints: Record<string, number>;
  vectors2: Record<string, number[]>; vectors3: Record<string, number[]>;
} {
  return material.serialize();
}

  it("uploads the local absolute source field once per changed physical state and releases resized resources", async () => {
    const physicalProfile = getEngineExhaustOpticalProfile("f135-visible-approximation-v1")!;
    const t = fixture({ opticalProfile: physicalProfile });
    const state = { ...running, augmentation: true, gasTemperatureKelvin: 2200, ambientTemperatureKelvin: 288.15 };
    t.handle.update(state);
    expect(t.handle.mesh.isEnabled()).toBe(false);
    await t.handle.ready;
    const field = () => t.scene.textures.find(texture => texture.name === "engine-exhaust-source-field") as RawTexture;
    const original = field();
    expect(original.getSize()).toEqual({ width: 64, height: 32 });
    expect(original.samplingMode).toBe(RawTexture.NEAREST_SAMPLINGMODE);
    expect(t.material.serialize().vectors4.fieldShape.slice(0, 2)).toEqual([64, 32]);
    expect(t.material.serialize().vectors4.fieldShape[3]).toBe(1);
    const uploaded = original.getInternalTexture()!._bufferView as Float32Array;
    const redFractions: number[] = [];
    for (let i = 0; i < uploaded.length; i += 4) {
      const sum = uploaded[i] + uploaded[i + 1] + uploaded[i + 2];
      if (sum > 1e-8) redFractions.push(uploaded[i] / sum);
    }
    // Actual uploaded RGB must vary in composition, not just one RGB times density.
    expect(Math.max(...redFractions) - Math.min(...redFractions)).toBeGreaterThan(0.1);
    const update = vi.spyOn(original, "update");
    const dispose = vi.spyOn(original, "dispose");
    t.requestRender.mockClear();
    t.handle.update({ ...state, simulationTimeSeconds: 20 });
    expect(update).not.toHaveBeenCalled();
    expect(t.requestRender).not.toHaveBeenCalled();
    t.handle.update({ ...state, augmentation: false });
    expect(update).not.toHaveBeenCalled();
    expect(t.requestRender).not.toHaveBeenCalled();
    t.handle.update(state, { ...settings, gasReferenceNits: 100, intensity: 2 });
    t.scene.imageProcessingConfiguration.exposure = 2;
    expect(update).not.toHaveBeenCalled();
    const hotter = { ...state, gasTemperatureKelvin: 2300 };
    t.handle.update(hotter);
    expect(update).toHaveBeenCalledOnce();
    t.handle.update(hotter);
    expect(update).toHaveBeenCalledOnce();
    t.handle.update(hotter, { ...settings, contributionView: "solid" });
    t.handle.update(state);
    expect(update).toHaveBeenCalledOnce();
    t.handle.update(state, settings);
    expect(update).toHaveBeenCalledTimes(2);
    t.handle.update(state, { ...settings, axialFieldSamples: 100, radialFieldSamples: 1 });
    expect(dispose).toHaveBeenCalledOnce();
    expect(field().getSize()).toEqual({ width: 64, height: 8 });
    expect(t.material.getActiveTextures()).not.toContain(original);
    expect(t.material.getActiveTextures()).toContain(field());
    const finalDispose = vi.spyOn(field(), "dispose");
    t.handle.dispose();
    expect(finalDispose).toHaveBeenCalledOnce();
  });

  it("keeps native-temperature hardware hot after combustion stops, then draws temperature-only cooling", async () => {
    const t = thermalFixture();
    await t.handle.ready;
    t.handle.update({ ...running, metalTemperatureKelvin: 1400 });
    const hot = t.emission();
    expect(hot.r).toBeCloseTo(t.restEmission.r + 0.4);
    expect(hot.g).toBeCloseTo(t.restEmission.g + 0.1);
    t.requestRender.mockClear();

    // Fuel cutoff stops the plume; the native temperature observation still
    // represents a hot engine. The renderer adds no cooling clock of its own.
    const stopped: EngineExhaustState = {
      ...running, running: false, augmentation: false, powerNorm: 0,
      simulationTimeSeconds: 11, metalTemperatureKelvin: 1400,
    };
    t.handle.update(stopped);
    expect(t.handle.mesh.isEnabled()).toBe(false);
    expect(t.emission().equals(hot)).toBe(true);
    expect(t.requestRender).toHaveBeenCalledOnce();
    t.requestRender.mockClear();

    // Everything used by the flame remains identical. A temperature-only
    // change must still update the existing material and request its frame.
    t.handle.update({ ...stopped, metalTemperatureKelvin: 1000 });
    expect(t.emission().r).toBeCloseTo(t.restEmission.r + 0.2);
    expect(t.emission().g).toBeCloseTo(t.restEmission.g + 0.05);
    expect(t.handle.mesh.isEnabled()).toBe(false);
    expect(t.requestRender).toHaveBeenCalledOnce();
    t.requestRender.mockClear();
    t.handle.update({ ...stopped, metalTemperatureKelvin: 600 });
    expect(t.emission().equals(t.restEmission)).toBe(true);
    expect(t.requestRender).toHaveBeenCalledOnce();
    t.requestRender.mockClear();
    t.handle.update({ ...stopped, metalTemperatureKelvin: 600 });
    expect(t.requestRender).not.toHaveBeenCalled();
  });

    expect(t.material.options.defines).toContain("#define CONTRAST");

    t.material.onBindObservable.notifyObservers(t.material);

    t.handle.dispose();
    t.requestRender.mockClear();

    expect(t.requestRender).not.toHaveBeenCalled();

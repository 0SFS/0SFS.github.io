  for(const view of ['rear','oblique'])for(const [x,y]of [[0,0],[.22,.1],[-.35,-.1],[.6,.4],[-.9,.2]]){const camera=cameraRay(view,x,y,input.radiusMeters,input.lengthMeters),reference=rayReference(f,input.radiusMeters,input.lengthMeters,camera.origin,camera.direction,4096),refined=rayReference(f,input.radiusMeters,input.lengthMeters,camera.origin,camera.direction,8192),refPeak=Math.max(...refined.radiance,1e-25),convergence=Math.max(...reference.radiance.map((v,c)=>Math.abs(v-refined.radiance[c])/refPeak));for(const samples of [4,8,16,32])for(const sampling of ['uniform','exit-clustered']){const actual=rayMidpoint(f,input.radiusMeters,input.lengthMeters,camera.origin,camera.direction,samples,sampling==='exit-clustered');rays.push({name,view,x,y,samples,sampling,radiance:actual.radiance,referenceRadiance:refined.radiance,absolutePeakErrorCdPerM2:Math.max(...actual.radiance.map((v,c)=>Math.abs(v-refined.radiance[c]))),relativeToReferencePeakError:Math.max(...actual.radiance.map((v,c)=>Math.abs(v-refined.radiance[c])/refPeak)),referenceConvergence:convergence});}assert(convergence<.002,`Reference transfer convergence ${name}/${view}/${x}`);}

// A constant source has the exact homogeneous solution; spatial variation is not
// intrinsically a decorative hue change. A deliberately nonuniform thermal field
// must change chromaticity without any hand-authored color ramp.

const dry=cases.find(c=>c.name==='powered-lift');

const uniformInputCheck={uniformPhysicalRanges,variablePhysicalRanges,description:'Identical temperature-retention/loading bases and zero chemical source, then an axial temperature-retention variation only; no color ramp or per-channel mask.'};
const causal=[];const onset=cases.find(c=>c.name==='ab-onset');for(const factor of [0,.25,.5,1]){const input={...onset.input,afterburnerBurnedFuelFlowKgPerSecond:onset.input.afterburnerBurnedFuelFlowKgPerSecond*factor},gas=evaluateEngineGasOptics(profile,input);causal.push({burnedFuelFactor:factor,excitedPowerW:gas.excitedPowerW,chPowerW:gas.chPowerW,c2PowerW:gas.c2PowerW});assert(Math.abs(gas.excitedPowerW-onset.gas.excitedPowerW*factor)<1e-9,'Band source not linear in actual burned fuel');}

const baselineManifestPath='validation/evidence/aircraft/f35b/engine-physical-correction-2026-10-06/optics/f135-exhaust-lut.manifest.json',oldSpatial=JSON.parse(read(baselineManifestPath)).spatialParameters;
function legacyUniformField(entry){
  const width=97,height=65,rgba=new Float32Array(width*height*4),native=entry.native,oldLength=native.lengthM,source=[native.sourceRcdM3,native.sourceGcdM3,native.sourceBcdM3];
  for(let y=0;y<height;y++)for(let x=0;x<width;x++){const oldZ=x/(width-1)*entry.input.lengthMeters/oldLength,radiusFraction=1-.6*oldZ,r=y/(height-1)/radiusFraction;let density=0;
    if(oldZ>=0&&oldZ<1&&r<1){const edge=Math.max(0,1-r*r)**2,annulus=Math.exp(-.5*((r-oldSpatial.ringRadiusFraction)/oldSpatial.ringWidthFraction)**2)*(1-oldSpatial.lobeContrast/2),ring=edge*Math.min(1,oldSpatial.centerEmission+(1-oldSpatial.centerEmission)*annulus),mix=Math.min(1,oldZ/oldSpatial.downstreamMixFraction);density=((1-mix)*ring+mix*edge)*(1-oldZ)**2;}
    const i=(y*width+x)*4;rgba.set([...source.map(v=>v*density),native.absorptionPerM*density],i);
  }
  return{width,height,rgba,radiusExpansionRatio:0};
}

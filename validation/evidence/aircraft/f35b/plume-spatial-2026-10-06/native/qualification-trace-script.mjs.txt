// The repository's deterministic encoder emits 8-bit RGBA, noninterlaced, filter 0.
const idat = []; let width, height;
for (let i=8;i<png.length;) {
  const size=png.readUInt32BE(i),name=png.toString('ascii',i+4,i+8),chunk=png.subarray(i+8,i+8+size);
  if(name==='IHDR'){width=chunk.readUInt32BE(0);height=chunk.readUInt32BE(4);assert.equal(chunk[8],8);assert.equal(chunk[9],6);assert.equal(chunk[12],0);}
  if(name==='IDAT')idat.push(chunk);i+=size+12;
}

for(let y=0;y<height;y++){assert.equal(raw[y*(width*4+1)],0);rgba.set(raw.subarray(y*(width*4+1)+1,(y+1)*(width*4+1)),y*width*4);}
const sampleMask=(x,y,channel)=>{
  x=Math.min(width-1,Math.max(0,x*(width-1)));y=Math.min(height-1,Math.max(0,y*(height-1)));
  const loX=Math.floor(x),loY=Math.floor(y),hiX=Math.min(width-1,loX+1),hiY=Math.min(height-1,loY+1),fx=x-loX,fy=y-loY;
  const p=(xx,yy)=>rgba[(yy*width+xx)*4+channel]/255;
  return (p(loX,loY)*(1-fx)+p(hiX,loY)*fx)*(1-fy)+(p(loX,hiY)*(1-fx)+p(hiX,hiY)*fx)*fy;
};
const density=(x,y,z)=>{
  const radius=1-.6*z,mix=Math.min(1,z/profile.spatialEmission.downstreamMixFraction);
  return ((1-mix)*sampleMask(x/radius*.5+.5,y/radius*.5+.5,0)+mix*sampleMask(x/radius*.5+.5,y/radius*.5+.5,1))*(1-z)**2;
};
const integral=(f,a,b,n)=>{let sum=0;for(let i=0;i<=n;i++)sum+=(i===0||i===n?.5:1)*f(a+(b-a)*i/n);return sum*(b-a)/n;};
const pathDensity=n=>({rearAxis:integral(z=>density(0,0,z),0,1,n),
  rearOffset:integral(z=>density(.55,0,z),0,1,n),sideNearExit:integral(x=>density(x,0,.05),-1,1,n)});

function optics(row){
  const geometry=helpers.nozzleApertureGeometry(row.nozzleNorm,helpers.F135_ENGINE_GEOMETRY.aperture);
  const length=6*(row.augmentation?.7+.3*Math.min(1,Math.max(0,row.n2Pct/100)):.2);
  const gas=helpers.evaluateEngineGasOptics(profile,{temperatureKelvin:row.gasK,augmentation:Boolean(row.augmentation),fuelFlowKgPerSecond:row.fuelKgSec,
    afterburnerBurnedFuelFlowKgPerSecond:row.burnedAbKgSec,radiusMeters:geometry.exitRadius,lengthMeters:length});
  const continuum=gas.blackbodyRgbCdPerM2.map(v=>v*gas.absorptionPerMeter);
  const excited=gas.sourceRgbCdPerM3.map((v,c)=>v-continuum[c]);
  Object.assign(row,{visualRunning:Number(row.n2Pct>0&&row.fuelKgSec>0),opticsValid:Number(gas.valid),radiusM:geometry.exitRadius,lengthM:length,
    throatAreaM2:geometry.throatArea,exitAreaM2:geometry.exitArea,absorptionPerM:gas.absorptionPerMeter,
    effectiveVolumeM3:gas.emittingVolumeM3,fuelPowerW:gas.fuelPowerW,particlePowerBoundW:gas.particlePowerUpperBoundW,
    excitedPowerW:gas.excitedPowerW,totalSourcePowerBoundW:gas.totalSourcePowerUpperBoundW,
    sourceYcdM3:luminance(gas.sourceRgbCdPerM3),continuumYcdM3:luminance(continuum),excitedYcdM3:luminance(excited)});
  for(const [c,name]of ['R','G','B'].entries()){row['source'+name+'cdM3']=gas.sourceRgbCdPerM3[c];row['continuum'+name+'cdM3']=continuum[c];row['excited'+name+'cdM3']=excited[c];}
  for(const [name,value]of Object.entries(densityIntegrals)){
    const effectivePath=value*(name==='sideNearExit'?geometry.exitRadius:length);
    const integral=gas.absorptionPerMeter>0?-Math.expm1(-gas.absorptionPerMeter*effectivePath)/gas.absorptionPerMeter:effectivePath;
    row[name+'EffectivePathM']=effectivePath;row[name+'RadianceYcdM2']=row.sourceYcdM3*integral;
    row[name+'SceneYAtWhite1000']=row[name+'RadianceYcdM2']/1000;
  }
  for(const solid of ['core','liner'])row[solid+'EmissionYcdM2']=luminance(helpers.interpolateAbsoluteEmission(profile.surfaceEmission,row[solid+'K'])??[0,0,0]);
  row.flowProxyKgSec=120*(row.n2Pct/100)**2*row.pressurePsf/2116.22*Math.sqrt(288.15/row.ambientK);
  row.directionX=-Math.sin(row.nozzlePitchRad)*Math.sin(row.nozzleYawRad);
  row.directionY=-Math.sin(row.nozzlePitchRad)*Math.cos(row.nozzleYawRad);row.directionZ=Math.cos(row.nozzlePitchRad);
  return row;
}
async function scenario(name,run){
  const sdk=await JSBSimSdk.create({moduleUrl:wasmModuleUrl,wasmUrl:wasmBinaryUrl,persistence:{enabled:false},log:{console:false}});
  sdk.on('stderr',({message})=>logs.push(name+': '+message));
  try{
    for(const[file,bytes]of files)sdk.writeDataFile(file,bytes.toString('utf8'));
    const initialOptions=helpers.engineTestStandBootstrapOptions('cold');await helpers.bootstrapAircraft(sdk,'f-35b',initialOptions);sdk.setDt(1/120);sdk.setHoldDown(true);
    const get=p=>sdk.getPropertyValue(p),set=(p,v)=>sdk.setPropertyValue(p,v),control=helpers.createEngineControl(sdk,helpers.getFdmProfile('f-35b'));
    propertyCatalog??=sdk.queryPropertyCatalog('');
    const propertyNames=new Set(propertyCatalog.split('\n').map(line=>line.trim().replace(/ \([^)]*\)$/,'')));
    for(const property of Object.values(mappings))assert.ok(propertyNames.has(property),'Missing required property '+property);
    const read=(phase,phaseTimeS=0)=>{const row={scenario:name,phase,timeS:sdk.getSimTime(),phaseTimeS,...Object.fromEntries(Object.entries(mappings).map(([key,p])=>[key,get(p)]))};row.ambientK+=273.15;row.fuelKgSec*=.45359237;return row;};
    let count=0;const initial=optics(read('initial-zero-time'));trace.push(initial);
    const step=(phase,phaseTimeS,startHeld=false,dense=false)=>{
      const before=sdk.getSimTime();control.step(startHeld);assert.equal(sdk.run(),true);const row=read(phase,phaseTimeS);
      assert.ok(Math.abs(row.timeS-before-1/120)<1e-9);assert.equal(row.thermalValid,1);
      assert.ok(row.burnedAbKgSec>=0&&row.burnedAbKgSec<=row.suppliedAbKgSec+1e-10&&row.suppliedAbKgSec<=row.fuelKgSec+1e-10);
      if(!row.augmentation)assert.equal(row.burnedAbKgSec,0);
      if(name==='powered-lift')assert.equal(row.augmentation,0,'No AB during dry powered lift, at every fixed step');
      auditedSteps++;if(++count%Math.round(sampleSeconds*120)===0||dense)trace.push(optics(row));return row;
    };
    const advance=(phase,seconds,throttle,denseSeconds=0)=>{if(throttle!==undefined)set('fcs/throttle-cmd-norm',throttle);let row;
      for(let i=0;i<Math.round(seconds*120);i++)row=step(phase,(i+1)/120,false,i<Math.round(denseSeconds*120)||i===0);
      optics(row);if(trace.at(-1)!==row)trace.push(row);return row;};
    const start=()=>{let row;for(let i=0;i<60*120;i++){row=step('starter',(i+1)/120,true);if(row.nativeRunning)break;}
      assert.equal(row.nativeRunning,1);optics(row);if(trace.at(-1)!==row)trace.push(row);return row;};
    const result=await run({sdk,get,set,control,read,step,advance,start});
    summaries.push({name,initialOptions,initial,result,final:optics(read('final'))});checks.push({name,passed:true});
  }catch(error){checks.push({name,passed:false,error:error.stack});failures.push(name+': '+error.message);}finally{sdk.destroy();}
}
await scenario('cold-cycle',({advance,start,control})=>{
  const cold=advance('cold-soak',5,0),started=start(),idle=advance('idle',60,0),dry=advance('dry99-before-ab',120,.99);
  const ab=advance('afterburner',60,1,3),returnedDry=advance('dry99-after-ab',120,.99,3);
  control.shutdown();const cutoff=advance('shutdown',1,0,1),cooled=advance('cooling',180,0);
  assert.equal(dry.augmentation,0);assert.equal(ab.augmentation,1);assert.equal(returnedDry.augmentation,0);assert.equal(cooled.fuelKgSec,0);
  return {cold,started,idle,dry,ab,returnedDry,cutoff,cooled};
});
await scenario('powered-lift',({sdk,step,start,set,advance,read})=>{
  step('cold-first-step',1/120);start();set('fcs/stovl-cmd-norm',1);const lifted=advance('powered-lift-full-command',90,1,5);
  assert.ok(lifted.conversion>.99);assert.ok(Math.abs(lifted.nozzlePitchRad-Math.PI/2)<1e-9);assert.equal(lifted.augmentation,0);assert.ok(lifted.opticsValid);
  const before=read('before-hold');sdk.hold();try{for(let i=0;i<50;i++)sdk.run();}finally{sdk.resume();}
  const after=read('after-hold');assert.deepEqual(after,{...before,phase:'after-hold'});trace.push(optics(after));
  return {lifted,holdIterations:50,holdStateExactlyPreserved:true};
});

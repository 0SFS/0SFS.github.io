#!/usr/bin/env node
// 0sfs owns aircraft engine geometry and its generated installed/full exports.
import { readFile,writeFile,mkdir } from 'node:fs/promises';
import path from 'node:path';
import { fileURLToPath } from 'node:url';
import { buildSource,profile } from './f135Engine/source.mjs';
import { exportEngine,removeOriginalNozzle,sha256 } from './f135Engine/glb.mjs';
import { newOutputDirectory } from './outputDirectory.mjs';
const root=fileURLToPath(new URL('../',import.meta.url));
const args=process.argv.slice(2),install=args.includes('--install'),check=args.includes('--check');
if(args.some(a=>!['--install','--check'].includes(a))||install&&check)throw new Error('Usage: node scripts/build-f135-engine.mjs [--install|--check]');
const source=buildSource(),assets=['full','installed'].map(variant=>({variant,path:`public/aircraft/f-35b/engine/F135-PW-600-${variant}.glb`,...exportEngine(source,variant)}));
const sourcePath='public/aircraft/f-35b/F-35B_AF267.glb',original=await readFile(path.join(root,sourcePath));
assets.push({variant:'airframe',path:'public/aircraft/f-35b/F-35B_AF267-airframe.glb',...removeOriginalNozzle(original)});
const inputs=await Promise.all(['scripts/build-f135-engine.mjs','scripts/f135Engine/source.mjs','scripts/f135Engine/profile.json','scripts/f135Engine/glb.mjs',sourcePath].map(async p=>({path:p,sha256:sha256(await readFile(path.join(root,p)))})));
const manifest={schema:'0sfs-f135-engine-assets/1',profile:profile.id,sourceInputs:inputs,referenceProvenance:profile.provenance,dimensions:profile.geometry,attachment:profile.attachment,variants:assets.map(({bytes,...asset})=>({...asset,bytes:bytes.length,sha256:sha256(bytes)})),textureCount:0,qualification:'Original approximation; geometry validation is separate from device or photometric qualification.'};
const manifestPath='public/aircraft/f-35b/engine/manifest.json';
const ts=`// Generated by scripts/build-f135-engine.mjs. Do not edit.\nexport const F135_ENGINE_GEOMETRY = ${JSON.stringify({profileId:profile.id,attachment:profile.attachment,...profile.geometry,assets:manifest.variants.map(({variant,path:assetPath,triangles,bytes,sha256})=>({variant,path:assetPath.replace(/^public\//,''),triangles,bytes,sha256}))},null,2)} as const;\n`;
const outputs=[...assets.map(a=>({path:a.path,bytes:a.bytes})),{path:manifestPath,bytes:Buffer.from(JSON.stringify(manifest,null,2)+'\n')},{path:'src/flight/aircraft/generated/f135EngineData.ts',bytes:Buffer.from(ts)}];
const out=check?null:newOutputDirectory('assets','f135-engine');
for(const output of outputs){if(check){const actual=await readFile(path.join(root,output.path));if(!actual.equals(output.bytes))throw new Error(`Stale artifact: ${output.path}`);}else{const target=install?path.join(root,output.path):path.join(out,path.basename(output.path));await mkdir(path.dirname(target),{recursive:true});await writeFile(target,output.bytes);}}
const report={mode:check?'check':install?'install':'generate',output:out?path.relative(root,out):null,artifacts:manifest.variants};
if(out)await writeFile(path.join(out,'generation.json'),JSON.stringify(report,null,2)+'\n');console.log(JSON.stringify(report,null,2));

import { createHash } from 'node:crypto';
export const sha256 = bytes => createHash('sha256').update(bytes).digest('hex');
export function encodeGlb(json, binary) {
  const jb=Buffer.from(JSON.stringify(json)),jl=Math.ceil(jb.length/4)*4,bl=Math.ceil(binary.length/4)*4;
  const bytes=Buffer.alloc(28+jl+bl);bytes.writeUInt32LE(0x46546c67);bytes.writeUInt32LE(2,4);bytes.writeUInt32LE(bytes.length,8);bytes.writeUInt32LE(jl,12);bytes.writeUInt32LE(0x4e4f534a,16);bytes.fill(32,20,20+jl);jb.copy(bytes,20);bytes.writeUInt32LE(bl,20+jl);bytes.writeUInt32LE(0x004e4942,24+jl);binary.copy(bytes,28+jl);return bytes;
}
export function decodeGlb(bytes) {
  if(bytes.readUInt32LE(0)!==0x46546c67||bytes.readUInt32LE(4)!==2||bytes.readUInt32LE(8)!==bytes.length)throw new Error('Invalid GLB');
  let json,binary;for(let i=12;i<bytes.length;){const length=bytes.readUInt32LE(i),type=bytes.readUInt32LE(i+4),chunk=bytes.subarray(i+8,i+8+length);if(type===0x4e4f534a)json=JSON.parse(chunk);if(type===0x004e4942)binary=chunk;i+=8+length;}return {json,binary};
}
export function accessor(gltf,index) {
  const a=gltf.json.accessors[index],v=gltf.json.bufferViews[a.bufferView],n={SCALAR:1,VEC2:2,VEC3:3,VEC4:4,MAT4:16}[a.type],w={5121:1,5123:2,5125:4,5126:4}[a.componentType],read={5121:'readUInt8',5123:'readUInt16LE',5125:'readUInt32LE',5126:'readFloatLE'}[a.componentType];
  if(a.sparse||!n||!w)throw new Error('Unsupported accessor');const values=[];for(let row=0;row<a.count;row++)for(let c=0;c<n;c++)values.push(gltf.binary[read]((v.byteOffset??0)+(a.byteOffset??0)+row*(v.byteStride??n*w)+c*w));return values;
}
export function exportEngine(source,variant) {
  const meshes=source.meshes.filter(m=>variant==='full'||!m.standOnly),binary=[],accessors=[],bufferViews=[];let offset=0;
  const data=values=>{const bytes=Buffer.alloc(values.length*4);values.forEach((v,i)=>bytes.writeFloatLE(v,i*4));const view=bufferViews.length;bufferViews.push({buffer:0,byteOffset:offset,byteLength:bytes.length,target:34962});binary.push(bytes);offset+=bytes.length;const min=[0,1,2].map(c=>Math.min(...values.filter((_,i)=>i%3===c))),max=[0,1,2].map(c=>Math.max(...values.filter((_,i)=>i%3===c)));accessors.push({bufferView:view,componentType:5126,count:values.length/3,type:'VEC3',min,max});return accessors.length-1;};
  const nodes=source.transforms.map(t=>{const {parent,...n}=t;return {...n,...(n.translation?{translation:[...n.translation]}:{}),children:[]};});
  const nodeMap=new Map(nodes.map((n,i)=>[n.name,i]));
  source.transforms.forEach((t,i)=>{if(t.parent)nodes[nodeMap.get(t.parent)].children.push(i);});
  const gm=meshes.map((m,i)=>{const primitive={attributes:{POSITION:data(m.positions),NORMAL:data(m.normals)},material:source.materials.findIndex(mat=>mat.name===m.material),mode:4};if(m.openPositions)primitive.targets=[{POSITION:data(m.openPositions.map((v,j)=>v-m.positions[j])),NORMAL:data(m.openNormals.map((v,j)=>v-m.normals[j]))}];nodes[nodeMap.get(m.parent)].children.push(nodes.length);nodes.push({name:m.name,mesh:i,extras:m.extras});return {name:m.name,primitives:[primitive],...(m.openPositions?{weights:[0],extras:{targetNames:['nozzleOpen']}}:{})};});
  const json={asset:{version:'2.0',generator:'0sfs original F135 source v1',copyright:'Original visual reconstruction, AGPL-3.0-only; no F135 calibration claim'},scene:0,scenes:[{nodes:[0]}],nodes,meshes:gm,materials:source.materials,accessors,bufferViews,buffers:[{byteLength:offset}],extras:{variant,profile:'f135-pw600-reconstruction-v1',axes:'+Y up; +Z aft',units:'metres',morphTarget:'nozzleOpen',approximation:true}};

}
export function removeOriginalNozzle(bytes) {
  const {json:original,binary}=decodeGlb(bytes),j=structuredClone(original),removed=new Set(), root=j.nodes.findIndex(n=>n.name==='vtol');
  if(root<0||j.skins?.length||j.animations?.length)throw new Error('Unexpected source hierarchy');
  // AF267's `extras` also includes one disconnected, obsolete static engine
  // tube (124 triangles). Keep its other components, materials and vertex bytes.
  // The original source remains unchanged; these guards pin the audited topology.
  const extras=j.nodes.find(n=>n.name==='extras'),tube=j.meshes[extras.mesh].primitives[1];
  const tubeIndices=accessor({json:original,binary},tube.indices);
  if(tubeIndices.length!==936||tubeIndices.slice(0,564).some(i=>i>=164)||tubeIndices.slice(564).some(i=>i<164))throw new Error('Legacy extras topology changed; re-audit before removing anything');
  for(const ai of Object.values(tube.attributes)){
    const a=j.accessors[ai],v=j.bufferViews[a.bufferView],components={VEC2:2,VEC3:3}[a.type];
    if(a.componentType!==5126||a.byteOffset||v.byteStride)throw new Error('Unexpected legacy extras vertex layout');
    a.count=164;v.byteLength=164*components*4;
    if(a.min||a.max){const values=accessor({json:j,binary},ai);a.min=Array.from({length:components},(_,k)=>Math.min(...values.filter((_,i)=>i%components===k)));a.max=Array.from({length:components},(_,k)=>Math.max(...values.filter((_,i)=>i%components===k)));}
  }
  j.accessors[tube.indices].count=564;j.bufferViews[j.accessors[tube.indices].bufferView].byteLength=564*2;

  const visit=i=>{removed.add(i);for(const child of j.nodes[i].children??[])visit(child);};visit(root);
  const mapping=new Map();j.nodes.forEach((_,i)=>{if(!removed.has(i))mapping.set(i,mapping.size);});
  j.nodes=j.nodes.flatMap((n,i)=>removed.has(i)?[]:[{...n,...(n.children?{children:n.children.filter(c=>!removed.has(c)).map(c=>mapping.get(c))}:{})}]);
  j.scenes.forEach(s=>s.nodes=s.nodes.filter(n=>!removed.has(n)).map(n=>mapping.get(n)));
  const meshMap=new Map();for(const n of j.nodes)if(n.mesh!==undefined&&!meshMap.has(n.mesh))meshMap.set(n.mesh,meshMap.size);
  j.meshes=[...meshMap.keys()].map(i=>j.meshes[i]);for(const n of j.nodes)if(n.mesh!==undefined)n.mesh=meshMap.get(n.mesh);
  const needed=new Set();for(const m of j.meshes)for(const p of m.primitives){for(const a of Object.values(p.attributes))needed.add(a);if(p.indices!==undefined)needed.add(p.indices);for(const t of p.targets??[])for(const a of Object.values(t))needed.add(a);}
  const am=new Map([...needed].sort((a,b)=>a-b).map((v,i)=>[v,i]));j.accessors=[...am.keys()].map(i=>j.accessors[i]);for(const m of j.meshes)for(const p of m.primitives){for(const k of Object.keys(p.attributes))p.attributes[k]=am.get(p.attributes[k]);if(p.indices!==undefined)p.indices=am.get(p.indices);for(const t of p.targets??[])for(const k of Object.keys(t))t[k]=am.get(t[k]);}
  const views=new Set(j.accessors.map(a=>a.bufferView));for(const i of j.images??[])if(i.bufferView!==undefined)views.add(i.bufferView);
  const vm=new Map([...views].sort((a,b)=>a-b).map((v,i)=>[v,i]));let size=0;const chunks=[];j.bufferViews=[...vm.keys()].map(i=>{const v=j.bufferViews[i],chunk=binary.subarray(v.byteOffset??0,(v.byteOffset??0)+v.byteLength),aligned=Buffer.alloc(Math.ceil(chunk.length/4)*4);chunk.copy(aligned);chunks.push(aligned);const next={...v,byteOffset:size,buffer:0};size+=aligned.length;return next;});for(const a of j.accessors)a.bufferView=vm.get(a.bufferView);for(const i of j.images??[])if(i.bufferView!==undefined)i.bufferView=vm.get(i.bufferView);j.buffers=[{byteLength:size}];j.asset.generator=`${j.asset.generator??'glTF'}; 0sfs remove rejected engine subtree`;
  const output=encodeGlb(j,Buffer.concat(chunks)),parsed=decodeGlb(output);

  for(const [old,n]of vm){const ov=original.bufferViews[old],nv=j.bufferViews[n];if(!binary.subarray(ov.byteOffset??0,(ov.byteOffset??0)+nv.byteLength).equals(parsed.binary.subarray(nv.byteOffset,nv.byteOffset+nv.byteLength)))throw new Error('Airframe bytes changed');}

}
export function posedMeshes(gltf,aperture=0) {
  return gltf.json.nodes.filter(n=>n.mesh!==undefined).map(node=>{const m=gltf.json.meshes[node.mesh],p=m.primitives[0],base=accessor(gltf,p.attributes.POSITION),delta=p.targets?.[0]?accessor(gltf,p.targets[0].POSITION):null,normalDelta=p.targets?.[0]?.NORMAL===undefined?null:accessor(gltf,p.targets[0].NORMAL),parents=gltf.json.nodes.filter(n=>n.children?.includes(gltf.json.nodes.indexOf(node)));return {name:node.name,parent:parents[0]?.name,positions:base.map((v,i)=>v+(delta?.[i]??0)*aperture),normals:accessor(gltf,p.attributes.NORMAL).map((v,i)=>v+(normalDelta?.[i]??0)*aperture),material:gltf.json.materials[p.material].name,standOnly:node.extras?.standOnly};});
}
